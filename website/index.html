      def is_prime(n):
    if n < 2:
        return False
    for i in range(2, int(n**0.5) + 1):
        if n % i == 0:
            return False
    return True

fib = [0, 1]
while len(fib) < 10:
    fib.append(fib[-1] + fib[-2])

print("First 10 Fibonacci Numbers & Primality:")
for idx, num in enumerate(fib, 1):
    status = "PRIME" if is_prime(num) else "Not prime"
    print(f"#{idx:02d}: {num:3d} -> {status}")
